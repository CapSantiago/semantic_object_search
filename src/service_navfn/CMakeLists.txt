cmake_minimum_required(VERSION 3.8)
project(service_navfn)
find_package(ament_cmake REQUIRED)
find_package(nav2_navfn_planner REQUIRED)
find_package(pluginlib REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
add_library(service_navfn SHARED src/planner.cpp src/odom_memory.cpp)
target_compile_features(service_navfn PUBLIC cxx_std_17)
target_include_directories(service_navfn PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(service_navfn nav2_navfn_planner pluginlib nav2_costmap_2d)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
pluginlib_export_plugin_description_file(nav2_costmap_2d obstacle_plugins.xml)
install(TARGETS service_navfn LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(cell_coordinates_test test/cell_coordinates.cpp)
  target_link_libraries(cell_coordinates_test service_navfn)
  ament_target_dependencies(cell_coordinates_test nav2_navfn_planner)
  add_test(NAME cell_coordinates COMMAND cell_coordinates_test)
  add_executable(odom_memory_test test/odom_memory.cpp)
  target_link_libraries(odom_memory_test service_navfn)
  ament_target_dependencies(odom_memory_test nav2_navfn_planner nav2_costmap_2d)
  add_test(NAME odom_memory COMMAND odom_memory_test)
  add_executable(depth_clearing_test test/depth_clearing.cpp)
  ament_target_dependencies(depth_clearing_test nav2_costmap_2d)
  add_test(NAME depth_clearing COMMAND depth_clearing_test)
endif()
ament_package()
